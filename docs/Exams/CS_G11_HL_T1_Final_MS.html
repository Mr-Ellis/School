10 Function definition with def find_max(numbers): (1). Handles empty list by returning None (1). Initialises a max_value variable (1). Loops through the list and updates max_value when a larger value is found (1). Returns max_value (1).
def find_max(numbers):
    if len(numbers) == 0:
        return None
    max_value = numbers[0]
    for num in numbers:
        if num > max_value:
            max_value = num
    return max_value
