def calculate_average(numbers):
    """Returns the average of a list of numbers, or None if empty."""
    if len(numbers) == 0:
        return None
    total = 0
    for num in numbers:
        total = total + num
    return total / len(numbers)

9 Defines a function named get_valid_grade (1).
Uses a while True loop to repeatedly ask for input (1).
Uses try-except to handle ValueError if the input is not an integer (2).
Checks the grade is between 0 and 100 (1).
Returns the valid grade (1). 
def get_valid_grade():
    while True:
        try:
            grade = int(input("Enter a grade (0-100): "))
            if 0 <= grade <= 100:
                return grade
            else:
                print("Grade must be between 0 and 100.")
        except ValueError:
            print("Please enter a valid integer.")
