💻 Code Examples

1. Static Array (Python)


class StaticArray:
    def __init__(self, size):
        self.size = size
        self.array = [None] * size
        self.count = 0

    def insert(self, index, value):
        if index < 0 or index >= self.size:
            raise IndexError("Index out of bounds")
        self.array[index] = value
        self.count += 1

    def access(self, index):
        if index < 0 or index >= self.size:
            raise IndexError("Index out of bounds")
        return self.array[index]

    def display(self):
        print(f"Static Array: {self.array}")

2. Dynamic Linked List (Python)


class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None
        self.size = 0

    def append(self, data):
        new_node = Node(data)
        if self.head is None:
            self.head = new_node
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_node
        self.size += 1

    def access(self, index):
        if index < 0 or index >= self.size:
            raise IndexError("Index out of bounds")
        current = self.head
        for _ in range(index):
            current = current.next
        return current.data

    def display(self):
        elements = []
        current = self.head
        while current:
            elements.append(current.data)
            current = current.next
        print(f"Linked List: {elements}")

3. Performance Comparison


import time

# Time access to the last element
def time_access_static(array, index, iterations=1000000):
    start = time.time()
    for _ in range(iterations):
        _ = array.array[index]
    return time.time() - start

def time_access_dynamic(linked_list, index, iterations=1000):
    start = time.time()
    for _ in range(iterations):
        _ = linked_list.access(index)
    return time.time() - start
