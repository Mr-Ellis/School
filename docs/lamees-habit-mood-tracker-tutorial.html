The Habit Class
SC1 — Store one record

Now we build the engine. A Habit class stores everything about one habit entry: the date, the habit name, the mood score, and whether the habit was completed.

⚠️ Important

The __init__.py file must exist even if empty. It tells Python to treat models as a package so you can import from it.

models/habit.py 
from datetime import datetime, date


class Habit:
    def __init__(self, habit_id, date_str, habit_name, mood, completed):
        self.habit_id = habit_id
        self.date = date_str
        self.habit_name = habit_name
        self.mood = int(mood)
        self.completed = bool(completed)

    def to_dict(self):
        return {
            "id": self.habit_id,
            "date": self.date,
            "habit_name": self.habit_name,
            "mood": self.mood,
            "completed": self.completed,
        }

    def __repr__(self):
        status = "✓" if self.completed else "✗"
        return f"<Habit {self.habit_name} {self.date} mood={self.mood} {status}>"

SQLite Setup
SC10 — Persistent storage

If your program closes, you don't want to lose all your habits. We save them to a SQLite database — a single file called habits.db.

Why SQLite instead of JSON? SQLite supports date-range queries (SC2), average calculations (SC4), and it's designed for exactly this kind of structured data. It's also built into Python — no installation needed.

Create these files:

Folder structure 
HabitTracker\
└── managers\
    ├── __init__.py     (empty)
    └── database.py
managers/database.py 
import sqlite3

DB_PATH = "habits.db"


class Database:
    def __init__(self, path=DB_PATH):
        self.conn = sqlite3.connect(path)
        self.conn.row_factory = sqlite3.Row
        self.create_table()

    def create_table(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS habits (
                id          INTEGER PRIMARY KEY AUTOINCREMENT,
                date        TEXT    NOT NULL,
                habit_name  TEXT    NOT NULL,
                mood        INTEGER NOT NULL,
                completed   INTEGER NOT NULL
            )
        """)
        .conn.commit()

    def insert(self, habit):
        cursor = self.conn.cursor()
        cursor.execute(
            "INSERT INTO habits (date, habit_name, mood, completed) VALUES (?, ?, ?, ?)",
            (habit.date, habit.habit_name, habit.mood, int(habit.completed))
        )
        .conn.commit()
        return cursor.lastrowid

    def get_all(self):
        return self.conn.execute("SELECT * FROM habits ORDER BY date DESC").fetchall()

    def get_between(self, start_date, end_date):
        return self.conn.execute(
            "SELECT * FROM habits WHERE date BETWEEN ? AND ? ORDER BY date DESC",
            (start_date, end_date)
        ).fetchall()

    def update(self, habit_id, habit):
        self.conn.execute(
            "UPDATE habits SET date = ?, habit_name = ?, mood = ?, completed = ? WHERE id = ?",
            (habit.date, habit.habit_name, habit.mood, int(habit.completed), habit_id)
        )
        .conn.commit()

    def delete(self, habit_id):
        self.conn.execute("DELETE FROM habits WHERE id = ?", (habit_id,))
        .conn.commit()
❓ Your Turn

Why store completed as 0/1 instead of True/False?

Answer: SQLite doesn't have a native boolean type — it stores booleans as integers (0 or 1). We convert with int() when writing and bool() when reading.

Rejecting Bad Input
SC7 — Validation

Users make mistakes. Your program should catch them before saving garbage data.

Create this file:

Folder structure 
utilities\
├── __init__.py     (empty)
└── validation.py
utilities/validation.py 
from datetime import datetime


class Validation:
    @staticmethod
    def validate_habit_entry(date_text, habit_name, mood, completed):
        # Date must parse as YYYY-MM-DD
        try:
            datetime.strptime(date_text, "%Y-%m-%d")
        except ValueError:
            raise ValueError("Date must be in YYYY-MM-DD format.")

        # Habit name must be non-empty and short
        if not habit_name or not habit_name.strip():
            raise ValueError("Habit name cannot be empty.")
        if len(habit_name.strip()) > 50:
            raise ValueError("Habit name must be 50 characters or fewer.")

        # Mood must be a whole number from 1 to 10
        try:
            mood_int = int(mood)
        except (TypeError, ValueError):
            raise ValueError("Mood must be a whole number.")
        if not 1 <= mood_int <= 10:
            raise ValueError("Mood must be between 1 and 10.")

        # Completion must be a boolean-ish value
        if completed not in (True, False, 0, 1):
            raise ValueError("Completion must be yes or no.")

        return {
            "date": date_text,
            "habit_name": habit_name.strip(),
            "mood": mood_int,
            "completed": bool(completed),
        }
❓ Your Turn

Why validate before saving rather than after?

Answer: If you save first, invalid data ends up in your database, and you'll have to detect and clean it later. Validating first prevents bad data from ever entering the system.

The Tracker
SC1, SC2, SC8 — CRUD

The HabitTracker is the brain. It ties together the Habit model, the database, and the analytics.

managers/habit_tracker.py 
from models.habit import Habit
from managers.database import Database


class HabitTracker:
    def __init__(self):
        self.db = Database()

    def add_habit(self, date_str, habit_name, mood, completed):
        habit = Habit(None, date_str, habit_name, mood, completed)
        habit_id = self.db.insert(habit)
        habit.habit_id = habit_id
        return habit

    def get_all_habits(self):
        rows = self.db.get_all()
        return [
            Habit(r["id"], r["date"], r["habit_name"],
                  r["mood"], bool(r["completed"]))
            for r in rows
        ]

    def search_by_date_range(self, start, end):
        rows = self.db.get_between(start, end)
        return [
            Habit(r["id"], r["date"], r["habit_name"],
                  r["mood"], bool(r["completed"]))
            for r in rows
        ]

    def edit_habit(self, habit_id, date_str, habit_name, mood, completed):
        habit = Habit(habit_id, date_str, habit_name, mood, completed)
        self.db.update(habit_id, habit)

    def delete_habit(self, habit_id):
        .db.delete(habit_id)
❓ Your Turn

Where does get_all_habits() get its data from?

Answer: From the SQLite database via self.db.get_all(), which runs a SELECT * FROM habits query. Each row is converted back into a Habit object.

Streak Algorithm
SC3 — Consecutive days

A streak is the number of consecutive days a habit was completed. This is the feature that neither Habitica nor Daylio does well.

⚠️ The Edge Case Everyone Gets Wrong

If Julia logged Study on Monday, Tuesday, Wednesday, and it's now Thursday morning before she's logged anything, a naive algorithm returns 0. But her real streak is 3 — she just hasn't logged today yet. Handle this or she'll be demotivated by a wrong number.

utilities/analytics.py 
from datetime import datetime, timedelta


class Analytics:
    @staticmethod
    def streak(records, habit_name, today=None):
        """Return the current consecutive-day streak for one habit."""
        today = today or datetime.today().date()

        # Collect dates where this habit was completed
        completed_dates = {
            datetime.strptime(r.date, "%Y-%m-%d").date()
            for r in records
            if r.habit_name == habit_name and r.completed
        }

        if not completed_dates:
            return 0

        # Start from today, or yesterday if today isn't logged yet
        check = today
        if check not in completed_dates:
            check = today - timedelta(days=1)

        streak = 0
        while check in completed_dates:
            streak += 1
            check -= timedelta(days=1)
        return streak

Test in main.py

Mood Average
SC4 — Average over a period

SC4 says "calculate the average mood score over a selected period of time". This is straightforward, but you have to handle the case where there are no records in that period.

Add to utilities/analytics.py:

utilities/analytics.py 
@staticmethod
def mood_average(records, start_date, end_date):
    """Return the average mood between two dates (inclusive),
    or None if there are no records in that range."""
    moods = [
        r.mood for r in records
        if start_date <= r.date <= end_date
    ]
    if not moods:
        return None
    return round(sum(moods) / len(moods), 2)
❓ Your Turn

What happens if you remove the if not moods: return None line?

Answer: Python raises a ZeroDivisionError because you'd be dividing by len(moods) == 0. The guard prevents a crash when a period has no records.

Trend Detection
SC6 — Improving or declining?

SC6 says "detect trends in mood and habit consistency to determine whether the user's progress is improving or declining". We compare this week's average mood to last week's.

utilities/analytics.py 
@staticmethod
def detect_trend(records, today=None):
    """Return 'improving', 'declining', 'stable', or 'not enough data'."""
    today = today or datetime.today().date()
    this_start = str(today - timedelta(days=6))
    this_end   = str(today)
    last_start = str(today - timedelta(days=13))
    last_end   = str(today - timedelta(days=7))

    this_avg = Analytics.mood_average(records, this_start, this_end)
    last_avg = Analytics.mood_average(records, last_start, last_end)

    if this_avg is None or last_avg is None:
        return "not enough data"

    diff = this_avg - last_avg
    if diff >= 0.5:
        return "improving"
    elif diff <= -0.5:
        return "declining"
    return "stable"
💡 Why 0.5?

A 0.5-point difference on a 10-point scale is meaningful without being noise. If you set the threshold to 0.1, tiny mood fluctuations would flip the trend back and forth every day.

❓ Your Turn

What does the trend return if the user only has records from this week?

Answer: "not enough data" — because last_avg is None with no records last week. This is the safe default; better than misleading the user with a fake trend.

Feedback Engine
SC5 — Personalised motivational feedback

SC5 says feedback must be based on the user's actual mood score, recent progress, and streak length. If the message is generic ("Great job!"), you've rebuilt the problem you set out to solve.

utilities/feedback.py 
class Feedback:
    @staticmethod
    def generate(mood_avg, streak, trend):
        parts = []

        # Reference the streak
        if streak >= 7:
            parts.append(f"🔥 {streak}-day streak — you're building real momentum.")
        elif streak >= 3:
            parts.append(f"Nice — {streak} days in a row. Keep it going.")
        else:
            parts.append("Try to log today — every streak starts with day one.")

        # Reference the mood average
        if mood_avg is None:
            parts.append("Log a few mood scores and I'll show you patterns.")
        elif mood_avg >= 8:
            parts.append(f"Your average mood this period is {mood_avg:.1f}/10 — strong.")
        elif mood_avg >= 5:
            parts.append(f"Mood average is {mood_avg:.1f}/10 — steady, with room to grow.")
        else:
            parts.append(f"Mood average is {mood_avg:.1f}/10 — be kind to yourself this week.")

        # Reference the trend
        if trend == "improving":
            parts.append("Trend: improving compared to last week.")
        elif trend == "declining":
            parts.append("Trend: down from last week — consider a lighter schedule.")
        elif trend == "stable":
            parts.append("Trend: holding steady.")

        return " ".join(parts)
❓ Your Turn

Why does this count as "personalised" feedback?

Answer: Because every sentence references a number from the user's own data — their actual streak, their actual mood average, their actual trend. No two users see the same message unless their data is identical.

gui/main_window.py 
import tkinter as tk
from tkinter import ttk, messagebox
from datetime import date, timedelta
import customtkinter as ctk

from gui.habit_dialog import HabitDialog
from managers.habit_tracker import HabitTracker
from utilities.validation import Validation
from utilities.analytics import Analytics
from utilities.feedback import Feedback


class MainWindow(ctk.CTk):
    def __init__(self):
        super().__init__()
        .title("Habit & Mood Tracker")
        .geometry("1000x650")
        self.tracker = HabitTracker()
        .create_widgets()
        .refresh()

    def create_widgets(self):
        ctk.CTkLabel(self, text="Habit & Mood Tracker",
                     font=("Arial", 24, "bold")).pack(pady=10)

        btn_frame = ctk.CTkFrame(self)
        btn_frame.pack(fill="x", padx=20)
        ctk.CTkButton(btn_frame, text="Add", command=self.add_habit).pack(side="left", padx=5, pady=10)
        ctk.CTkButton(btn_frame, text="Edit", command=self.edit_habit).pack(side="left", padx=5)
        ctk.CTkButton(btn_frame, text="Delete", command=self.delete_habit).pack(side="left", padx=5)

        columns = ("Date", "Habit", "Mood", "Completed")
        self.tree = ttk.Treeview(self, columns=columns, show="headings")
        for col in columns:
            self.tree.heading(col, text=col)
            self.tree.column(col, width=200)
        self.tree.pack(fill="both", expand=True, padx=20, pady=20)

        self.feedback_label = ctk.CTkLabel(
            self, text="", wraplength=900,
            font=("Arial", 14), justify="left"
        )
        .feedback_label.pack(padx=20, pady=10, fill="x")

    def refresh(self):
        for row in self.tree.get_children():
            .tree.delete(row)

        records = .tracker.get_all_habits()
        for h in records:
            .tree.insert("", tk.END, iid=h.habit_id,
                         values=(h.date, h.habit_name, h.mood,
                                 "✓" if h.completed else "✗"))

        # Pick the most common habit name for the summary
        if records:
            habit_name = records[0].habit_name
            streak = Analytics.streak(records, habit_name)
            today = date.today()
            week_ago = str(today - timedelta(days=6))
            avg = Analytics.mood_average(records, week_ago, str(today))
            trend = Analytics.detect_trend(records)
            message = Feedback.generate(avg, streak, trend)
            .feedback_label.configure(text=f"💬 {message}")
        else:
            .feedback_label.configure(text="Add your first habit to see personalised feedback.")

    def add_habit(self):
        dialog = HabitDialog(self, "Add Habit")
        .wait_window(dialog)
        if not dialog.result:
            return

        date_str, name, mood, completed = dialog.result
        try:
            Validation.validate_habit_entry(date_str, name, mood, completed)
        except ValueError as err:
            messagebox.showerror("Error", str(err))
            return

        .tracker.add_habit(date_str, name, mood, completed)
        .refresh()

    def edit_habit(self):
        sel = .tree.selection()
        if not sel:
            messagebox.showwarning("Warning", "Select a record first.")
            return

        habit_id = int(sel[0])
        record = next((h for h in .tracker.get_all_habits() if h.habit_id == habit_id), None)

        dialog = HabitDialog(self, "Edit Habit", record)
        .wait_window(dialog)
        if not dialog.result:
            return

        date_str, name, mood, completed = dialog.result
        try:
            Validation.validate_habit_entry(date_str, name, mood, completed)
        except ValueError as err:
            messagebox.showerror("Error", str(err))
            return

        .tracker.edit_habit(habit_id, date_str, name, mood, completed)
        .refresh()

    def delete_habit(self):
        sel = .tree.selection()
        if not sel:
            messagebox.showwarning("Warning", "Select a record first.")
            return

        if messagebox.askyesno("Delete", "Delete this record?"):

@staticmethod
def mood_chart(records, days=14):
    """Return a simple text chart of the last N days of mood."""
    today = date.today()
    lines = []
    for i in range(days - 1, -1, -1):
        d = str(today - timedelta(days=i))
        day_moods = [r.mood for r in records if r.date == d]
        if day_moods:
            avg = sum(day_moods) / len(day_moods)
            bar = "█" * int(round(avg))
            lines.append(f"{d[5:]}  {bar} {avg:.1f}")
        else:
            lines.append(f"{d[5:]}  (no data)")
    return "\n".join(lines)
